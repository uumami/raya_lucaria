# Reader Shell Parity Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the no-top-bar reader shell official, tested, render-debug-safe, and accessible while preserving discovery workspace command bars and accepted comfort behavior.

**Architecture:** Keep the current Glintstone static renderer shape: reader pages use the structural left course rail, article, and right learning rail; discovery workspaces keep their command bars. Update authority docs and render-debug first, then port only mobile drawer accessibility behavior from `main-like-reader-shell` without its storage or breakpoint behavior. Tests should prove reader/discovery separation, storage policy, desktop geometry, mobile modal behavior, and render-debug parity.

**Tech Stack:** Python 3.10, `uv`, pytest, Playwright/Chromium, static HTML/CSS/JS generated by `packages/static`, render-debug reporting in `packages/cli`.

```python
def test_reader_shell_no_top_bar_geometry_across_desktop_viewports(
    tmp_path: Path,
) -> None:
    from playwright.sync_api import sync_playwright
    from raya_cli.preview import create_preview

    course = tmp_path / "render-fixture"
    shutil.copytree(RENDER_FIXTURE, course, ignore=shutil.ignore_patterns("artifact"))
    browser_executable = _browser_executable()

    handle = create_preview(course, host="127.0.0.1", port=0, dry_run=False)
    try:
        assert handle.report.ok, [
            diagnostic.format() for diagnostic in handle.report.diagnostics
        ]
        assert handle.base_url is not None
        with sync_playwright() as playwright:
            browser = playwright.chromium.launch(
                executable_path=str(browser_executable),
                headless=True,
                args=["--no-sandbox"],
            )
            try:
                for width in (1280, 1366, 1440, 1920):
                    page = browser.new_page(viewport={"width": width, "height": 900})
                    try:
                        page.goto(
                            f"{handle.base_url}/reader-ux/index.html",
                            wait_until="networkidle",
                        )
                        _assert_no_horizontal_overflow(page)
                        state = page.evaluate(
                            """() => {
                              const box = (selector) => {
                                const rect = document.querySelector(selector)
                                  .getBoundingClientRect();
                                return {
                                  left: rect.left,
                                  right: rect.right,
                                  width: rect.width,
                                  top: rect.top,
                                  bottom: rect.bottom,
                                };
                              };
                              const articleBefore = box("#raya-article");
                              document
                                .querySelector("[data-raya-course-map-toggle]")
                                .click();
                              return new Promise((resolve) => {
                                requestAnimationFrame(() => requestAnimationFrame(() => {
                                  const articleAfter = box("#raya-article");
                                  resolve({
                                    topBars: document
                                      .querySelectorAll(".raya-top-command-bar").length,
                                    map: box("#raya-course-map"),
                                    articleBefore,
                                    articleAfter,
                                    rail: box("#raya-learning-rail"),
                                    rootMapState: document.documentElement
                                      .dataset.rayaCourseMap,
                                    overflow: document.documentElement.scrollWidth -
                                      document.documentElement.clientWidth,
                                  });
                                }));
                              });
                            }"""
                        )
                        assert state["topBars"] == 0
                        assert state["map"]["right"] <= state["articleAfter"]["left"]
                        assert (
                            state["articleAfter"]["right"] <= state["rail"]["left"] + 1
                        )
                        assert (
                            state["articleAfter"]["width"]
                            >= state["articleBefore"]["width"] + 80
                        )
                        assert state["rootMapState"] == "collapsed"
                        assert state["overflow"] <= 1
                    finally:
                        page.close()
            finally:
                browser.close()
    finally:
        handle.close()
```
